bi new myapp                       # create a new project folder
cd myapp
bi serve src/main.bi               # start the web server on :8080
bi run src/main.bi                 # run without starting a server
bi init                            # add bi.json to current folder
bi install router                  # install a package
bi list                            # show installed packages
bi remove router                   # uninstall
bi help                            # show all commands
bi version                         # show version
